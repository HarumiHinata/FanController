cmake_minimum_required(VERSION 3.24)
project(FanController VERSION 0.3.1 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /utf-8 /permissive- /EHsc)
option(FAN_ASAN "Enable AddressSanitizer" OFF)
if(FAN_ASAN)
  add_compile_options(/fsanitize=address /Zi)
  add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(fan_core src/core/controller.cpp)
target_include_directories(fan_core PUBLIC src)
add_library(fan_history src/monitor/history.cpp)
target_include_directories(fan_history PUBLIC src)
add_library(fan_monitor src/monitor/sensors.cpp)
target_include_directories(fan_monitor PUBLIC src)
target_link_libraries(fan_monitor PUBLIC fan_history pdh dxgi gdi32)
add_library(fan_windows src/hardware/lenovo.cpp src/platform/platform.cpp src/platform/memory_optimizer.cpp)
target_include_directories(fan_windows PUBLIC src)
target_link_libraries(fan_windows PUBLIC fan_core advapi32 shell32 ole32 oleaut32 taskschd)
add_executable(FanProbe tools/probe.cpp)
target_link_libraries(FanProbe PRIVATE fan_windows)
add_executable(SensorProbe tools/sensor_probe.cpp)
target_link_libraries(SensorProbe PRIVATE fan_monitor)
add_executable(FanController WIN32 src/app/main.cpp src/app/dashboard.cpp resources/app.rc)
target_link_libraries(FanController PRIVATE fan_windows fan_monitor gdiplus comctl32 dwmapi psapi)
target_link_options(FanController PRIVATE /MANIFEST:NO)
enable_testing()
add_executable(CoreTests tests/core_tests.cpp)
target_link_libraries(CoreTests PRIVATE fan_core)
add_test(NAME core COMMAND CoreTests)
set_tests_properties(core PROPERTIES TIMEOUT 120)
add_executable(MonitorTests tests/monitor_tests.cpp)
target_link_libraries(MonitorTests PRIVATE fan_history)
add_test(NAME monitor COMMAND MonitorTests)
set_tests_properties(monitor PROPERTIES TIMEOUT 30)
install(TARGETS FanController RUNTIME DESTINATION .)
install(FILES README.md third_party/notices/Lenovo-Fan-Control.txt third_party/notices/AMD-ADL.txt third_party/notices/PCL2-Memory-Optimization.txt DESTINATION .)
install(FILES packaging/Install.ps1 packaging/Uninstall.ps1 DESTINATION .)
install(FILES docs/verification-report.md DESTINATION . RENAME verification-report.md)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "FanController-0.3.1-win64")
include(CPack)
